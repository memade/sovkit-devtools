cmake_minimum_required(VERSION 3.25)
project(sovkit-devtools VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
find_package(Python3 3.8 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
set(SOVKIT_SDK_ROOT "$ENV{SOVKIT_SDK_ROOT}" CACHE PATH "Unpacked SDK: include/sovkit.h, lib/libsovkit.*, README.md")
option(DEVTOOLS_BUILD_GUI "Build the wxWidgets developer workbench" ON)
option(DEVTOOLS_BUNDLE_SDK "Copy SDK runtime beside the application" ON)
if(NOT EXISTS "${SOVKIT_SDK_ROOT}/include/sovkit.h")
  message(FATAL_ERROR "Set SOVKIT_SDK_ROOT to a standalone SDK package, never an SDK source tree.")
endif()
if(WIN32)
  set(SOVKIT_RUNTIME_NAME libsovkit.dll)
elseif(APPLE)
  set(SOVKIT_RUNTIME_NAME libsovkit.dylib)
else()
  set(SOVKIT_RUNTIME_NAME libsovkit.so)
endif()
# Windows CMake installs use bin/, release SDK packages use lib/.
# Never search PATH or fall back to a library outside the selected SDK root.
set(SOVKIT_RUNTIME "${SOVKIT_SDK_ROOT}/lib/${SOVKIT_RUNTIME_NAME}")
if(WIN32 AND NOT EXISTS "${SOVKIT_RUNTIME}")
  set(SOVKIT_RUNTIME "${SOVKIT_SDK_ROOT}/bin/${SOVKIT_RUNTIME_NAME}")
endif()
if(NOT EXISTS "${SOVKIT_RUNTIME}")
  message(FATAL_ERROR "Missing target-platform shared library: ${SOVKIT_RUNTIME}")
endif()
set(SOVKIT_SDK_DOCUMENTATION "${SOVKIT_SDK_ROOT}/README.md")
if(NOT EXISTS "${SOVKIT_SDK_DOCUMENTATION}")
  message(FATAL_ERROR "SDK integration documentation is required")
endif()
set(generated "${CMAKE_CURRENT_BINARY_DIR}/generated")
file(MAKE_DIRECTORY "${generated}")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS
  "${SOVKIT_SDK_ROOT}/include/sovkit.h" scripts/generate_api.py scripts/embed_assets.py res/banner.xml "${SOVKIT_SDK_DOCUMENTATION}")
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/generate_api.py"
  "${SOVKIT_SDK_ROOT}/include/sovkit.h" "${generated}" COMMAND_ERROR_IS_FATAL ANY)
add_library(devtools_core STATIC src/sdk.cpp src/catalogue.cpp)
target_include_directories(devtools_core PUBLIC src "${generated}" "${SOVKIT_SDK_ROOT}/include")
target_link_libraries(devtools_core PUBLIC nlohmann_json::nlohmann_json Threads::Threads ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(devtools_core PUBLIC advapi32)
endif()
if(MSVC)
  target_compile_options(devtools_core PUBLIC /utf-8 /W4 /permissive-)
  target_compile_definitions(devtools_core PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)
else()
  target_compile_options(devtools_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(sovkit-console src/cli.cpp)
target_link_libraries(sovkit-console PRIVATE devtools_core)
install(TARGETS sovkit-console RUNTIME DESTINATION .)

if(DEVTOOLS_BUILD_GUI)
  find_package(wxWidgets CONFIG QUIET COMPONENTS core base xml aui html)
  if(NOT TARGET wx::core)
    find_package(wxWidgets REQUIRED COMPONENTS core base xml aui html)
    include(${wxWidgets_USE_FILE})
    add_library(devtools_wx INTERFACE)
    target_include_directories(devtools_wx INTERFACE ${wxWidgets_INCLUDE_DIRS})
    target_link_libraries(devtools_wx INTERFACE ${wxWidgets_LIBRARIES})
    target_compile_definitions(devtools_wx INTERFACE ${wxWidgets_DEFINITIONS})
  else()
    add_library(devtools_wx INTERFACE)
    target_link_libraries(devtools_wx INTERFACE wx::core wx::base wx::xml wx::aui wx::html)
  endif()
  find_package(fmt CONFIG REQUIRED)
  find_path(DEVTOOLS_RAPIDJSON_INCLUDE rapidjson/document.h REQUIRED)
  find_path(DEVTOOLS_UTFCPP_INCLUDE utf8.h PATH_SUFFIXES utf8cpp utfcpp REQUIRED)
  add_subdirectory(3rdparty/libwxui)
  execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed_assets.py"
    "${CMAKE_CURRENT_SOURCE_DIR}/res/banner.xml" "${SOVKIT_SDK_DOCUMENTATION}" "${generated}/banner.hpp"
    COMMAND_ERROR_IS_FATAL ANY)
  add_executable(sovkit-devtools WIN32 MACOSX_BUNDLE src/app.cpp)
  target_link_libraries(sovkit-devtools PRIVATE devtools_core libwxui)
  set_target_properties(sovkit-devtools PROPERTIES
    MACOSX_BUNDLE_BUNDLE_NAME "SovKit DevTools"
    MACOSX_BUNDLE_GUI_IDENTIFIER "com.skstu.sovkit.devtools"
    MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}"
    MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
  if(DEVTOOLS_BUNDLE_SDK)
    if(APPLE)
      set(runtime_destination "$<TARGET_BUNDLE_CONTENT_DIR:sovkit-devtools>/Frameworks")
    else()
      set(runtime_destination "$<TARGET_FILE_DIR:sovkit-devtools>")
    endif()
    add_custom_target(devtools-stage-sdk ALL
      COMMAND ${CMAKE_COMMAND} -E make_directory "${runtime_destination}"
      COMMAND ${CMAKE_COMMAND} -E copy_if_different "${SOVKIT_RUNTIME}" "${runtime_destination}/${SOVKIT_RUNTIME_NAME}"
      DEPENDS sovkit-devtools "${SOVKIT_RUNTIME}" VERBATIM)
  endif()
  install(TARGETS sovkit-devtools BUNDLE DESTINATION . RUNTIME DESTINATION .)
endif()
if(DEVTOOLS_BUNDLE_SDK)
  install(FILES "${SOVKIT_RUNTIME}" DESTINATION .)
  if(EXISTS "${SOVKIT_SDK_ROOT}/licenses")
    install(DIRECTORY "${SOVKIT_SDK_ROOT}/licenses/" DESTINATION licenses/sdk)
  endif()
  if(EXISTS "${SOVKIT_SDK_ROOT}/third_party")
    install(DIRECTORY "${SOVKIT_SDK_ROOT}/third_party/" DESTINATION third_party)
  endif()
  foreach(notice LICENSE NOTICE.md)
    if(EXISTS "${SOVKIT_SDK_ROOT}/${notice}")
      install(FILES "${SOVKIT_SDK_ROOT}/${notice}" DESTINATION licenses/sdk)
    endif()
  endforeach()
endif()
install(FILES LICENSE NOTICE.md README.md README.zh-CN.md CONTRIBUTING.md CHANGELOG.md DESTINATION .)
install(FILES 3rdparty/libwxui/LICENSE DESTINATION licenses/tools/libwxui)
install(FILES "${SOVKIT_SDK_DOCUMENTATION}" DESTINATION docs RENAME SDK_INTEGRATION.md)
install(DIRECTORY docs/ DESTINATION docs)
if(DEFINED VCPKG_INSTALLED_DIR AND DEFINED VCPKG_TARGET_TRIPLET)
  list(APPEND DEVTOOLS_NOTICE_ROOTS "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}")
endif()
list(APPEND DEVTOOLS_NOTICE_ROOTS ${CMAKE_PREFIX_PATH})
foreach(notice_root IN LISTS DEVTOOLS_NOTICE_ROOTS)
  file(GLOB notices "${notice_root}/share/*/copyright")
  foreach(notice IN LISTS notices)
    get_filename_component(port_dir "${notice}" DIRECTORY)
    get_filename_component(port "${port_dir}" NAME)
    install(FILES "${notice}" DESTINATION "licenses/tools/${port}")
  endforeach()
endforeach()
if(BUILD_TESTING)
  add_executable(devtools-core-tests tests/core_test.cpp)
  target_link_libraries(devtools-core-tests PRIVATE devtools_core)
  add_test(NAME devtools.core COMMAND devtools-core-tests "${SOVKIT_RUNTIME}")
  set_tests_properties(devtools.core PROPERTIES TIMEOUT 40)
  add_test(NAME devtools.two_peers COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/two_peers.py"
    --console "$<TARGET_FILE:sovkit-console>" --library "${SOVKIT_RUNTIME}")
  set_tests_properties(devtools.two_peers PROPERTIES TIMEOUT 120)
endif()
set(CPACK_PACKAGE_NAME SovKit-DevTools)
set(CPACK_PACKAGE_VERSION "${PROJECT_VERSION}")
set(CPACK_PACKAGE_FILE_NAME "SovKit-DevTools-${PROJECT_VERSION}-${CMAKE_SYSTEM_NAME}-${CMAKE_SYSTEM_PROCESSOR}-${CMAKE_BUILD_TYPE}")
set(CPACK_GENERATOR ZIP)
include(CPack)
